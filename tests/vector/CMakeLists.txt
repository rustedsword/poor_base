    target_compile_options(poor_vector_fortify${level}_tests PRIVATE -O2 -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=${level}

    endforeach()
endforeach()

function(vector_compile_test case regex)
    add_library(poor_vector_compile_${case} OBJECT EXCLUDE_FROM_ALL poor_vector_compile_${case}.c)
    target_link_libraries(poor_vector_compile_${case} poor_base)
    add_test(NAME poor_vector_compile_${case} COMMAND ${CMAKE_COMMAND} --build . --target poor_vector_compile_${case} WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
    set_tests_properties(poor_vector_compile_${case} PROPERTIES PASS_REGULAR_EXPRESSION "${regex}")
endfunction()
